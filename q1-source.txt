   138	static func aggregate_combos(summaries: Array) -> Array:
   139		## FRM-SYS23-09 — per combo_index: MET avg／min／max + combo_pass.
   140		## Aborted runs excluded from avg／min／max; all-aborted → blanks (not 0).
   141		var by_combo: Dictionary = {}
   142		for s_v in summaries:
   143			var s: Dictionary = s_v
   144			var ci: int = _json_int(s["combo_index"])
   145			if not by_combo.has(ci):
   146				by_combo[ci] = []
   147			(by_combo[ci] as Array).append(s)
   148		var keys: Array = by_combo.keys()
   149		keys.sort()
   150		var rows: Array = []
   151		for ci_v in keys:
   152			var ci2: int = _json_int(ci_v)
   153			var group: Array = by_combo[ci2]
   154			var combo_s: String = String((group[0] as Dictionary)["combo"])
   155			var row := {
   156				"combo_index": ci2,
   157				"combo": combo_s,
   158			}
   159			var all_pass := true
   160			var any_na := false
   161			var any_alive := false
   162			for met_id in ["MET-01", "MET-02", "MET-03", "MET-04", "MET-05a", "MET-05b", "MET-06"]:
   163				var vals: Array = []
   164				for g in group:
   165					var gd: Dictionary = g as Dictionary
   166					if _json_int(gd.get("aborted", 0)) == 1:
   167						continue
   168					any_alive = true
   169					var raw: Variant = gd[met_id]
   170					if typeof(raw) == TYPE_STRING and String(raw) == "":
   171						continue
   172					vals.append(float(raw))
   173				if vals.is_empty():
   174					row["%s_avg" % met_id] = ""
   175					row["%s_min" % met_id] = ""
   176					row["%s_max" % met_id] = ""
   177				else:
   178					row["%s_avg" % met_id] = _avg(vals)
   179					row["%s_min" % met_id] = _minv(vals)
   180					row["%s_max" % met_id] = _maxv(vals)
   181				## D-93／WP-27: MET-05b never judges pass — exclude from combo_pass.
   182				if met_id == "MET-05b":
   183					continue
   184				for g2 in group:
   185					var gd2: Dictionary = g2 as Dictionary
   186					if _json_int(gd2.get("aborted", 0)) == 1:
   187						continue
   188					var p: String = String(gd2.get("%s_pass" % met_id, PASS_NA))
   189					if p == PASS_NA:
   190						any_na = true
   191					elif p != "1":
   192						all_pass = false
   193			if not any_alive:
   194				row["combo_pass"] = PASS_NA
   195			elif any_na:
   196				row["combo_pass"] = PASS_NA
   197			elif all_pass:
   198				row["combo_pass"] = "1"
   199			else:
   200				row["combo_pass"] = "0"
   201			rows.append(row)
   202		return rows

   211		## 9.8 分母＝weekly_offtake>0 的 (港,貨)×評估季格數；無樣本 → 空白（D-52c）。

   229	static func _met02(cells: Array, th: Dictionary, seasons_y: int) -> Dictionary:
   230		## 9.8 分母＝可算比值的 (港,貨) 格（第一評估年平均週數≠0）；無樣本 → 空白（D-52c）。
   231		## D-93: stock_weeks from MARKET only (sampler／MarketPricer already MARKET).
   232		## Group by port|good; first eval year avg weeks vs last year.
   233		var by_cell: Dictionary = {}
   234		for c_v in cells:
   235			var c: Dictionary = c_v
   236			if String(c["stock_weeks"]) == "":
   237				continue
   238			var k: String = "%s|%s" % [c["port_id"], c["good_id"]]
   239			if not by_cell.has(k):
   240				by_cell[k] = []
   241			(by_cell[k] as Array).append(c)
   242		var diverged: int = 0
   243		var diverged_keys: Array = []
   244		var sampled: int = 0
   245		var th02: Variant = th.get("TH-02", null)
   246		var cell_keys: Array = by_cell.keys()
   247		cell_keys.sort()
   248		for k_v in cell_keys:
   249			var rows: Array = by_cell[k_v]
   250			rows.sort_custom(func(a: Dictionary, b: Dictionary) -> bool:
   251				return _json_int(a["season_index"]) < _json_int(b["season_index"])
   252			)
   253			if rows.is_empty():
   254				continue
   255			var first_si: int = _json_int((rows[0] as Dictionary)["season_index"])
   256			var last_si: int = _json_int((rows[rows.size() - 1] as Dictionary)["season_index"])
   257			var first_year_end: int = first_si + seasons_y
   258			var last_year_start: int = last_si - seasons_y + 1
   259			var first_sum: float = 0.0
   260			var first_n: int = 0
   261			var last_sum: float = 0.0
   262			var last_n: int = 0
   263			for r_v in rows:
   264				var r: Dictionary = r_v
   265				var si: int = _json_int(r["season_index"])
   266				var w: float = float(r["stock_weeks"])
   267				if si < first_year_end:
   268					first_sum = first_sum + w
   269					first_n = first_n + 1
   270				if si >= last_year_start:
   271					last_sum = last_sum + w
   272					last_n = last_n + 1
   273			var first_avg: float = 0.0
   274			if first_n > 0:
   275				first_avg = first_sum / float(first_n)
   276			var last_avg: float = 0.0
   277			if last_n > 0:
   278				last_avg = last_sum / float(last_n)
   279			if first_avg == 0.0:
   280				continue ## 「由零起」— excluded from > TH-02 count per 9.8
   281			sampled = sampled + 1
   282			var ratio: float = last_avg / first_avg
   283			if th02 != null and ratio > float(th02):
   284				diverged = diverged + 1
   285				diverged_keys.append(String(k_v))
   286		if sampled == 0:
   287			return {"value": "", "pass": PASS_NA, "diverged_keys": []}
   288		var pass_s := PASS_NA
   289		if th.has("TH-02"):
   290			pass_s = "1" if diverged == 0 else "0"
   291		return {"value": float(diverged), "pass": pass_s, "diverged_keys": diverged_keys}

   294	static func _met03(ports: Array, th: Dictionary, seasons_y: int) -> Dictionary:
   295		## 9.8 分母＝最後一年第一季總人口>0 的港；無樣本 → 空白（D-52c）。
   296		var by_port: Dictionary = _group_ports(ports)
   297		var worst: float = 0.0
   298		var sampled: int = 0
   299		for pid_v in by_port.keys():
   300			var rows: Array = by_port[pid_v]
   301			rows.sort_custom(func(a: Dictionary, b: Dictionary) -> bool:
   302				return _json_int(a["season_index"]) < _json_int(b["season_index"])
   303			)
   304			if rows.is_empty():
   305				continue
   306			var last: Dictionary = rows[rows.size() - 1]
   307			var last_si: int = _json_int(last["season_index"])
   308			var year_start_si: int = last_si - seasons_y + 1
   309			var first_of_last_year: Dictionary = last
   310			for r_v in rows:
   311				var r: Dictionary = r_v
   312				if _json_int(r["season_index"]) == year_start_si:
   313					first_of_last_year = r
   314					break
   315			var pop_last: int = _total_pop(last)
   316			var pop_first: int = _total_pop(first_of_last_year)
   317			if pop_first <= 0:
   318				continue
   319			sampled = sampled + 1
   320			var delta: float = absf(float(pop_last - pop_first)) / float(pop_first)
   321			if delta > worst:
   322				worst = delta
   323		if sampled == 0:
   324			return {"value": "", "pass": PASS_NA}
   325		return _pass_le(worst, th, "TH-03")

   370	static func _met05(ports: Array, th: Dictionary, seasons_y: int) -> Dictionary:
   371		## 9.8 分母＝有季表的 CITY／港；無港樣本 → a／b 空白（D-52c）。實測 0 保留。
   372		var by_port: Dictionary = _group_ports(ports)
   373		if by_port.is_empty():
   374			return {"a": "", "a_pass": PASS_NA, "b": "", "b_pass": PASS_NA}
   375		var longest_zero: int = 0
   376		var worst_growth: float = 0.0
   377		for pid_v in by_port.keys():
   378			var rows: Array = by_port[pid_v]
   379			rows.sort_custom(func(a: Dictionary, b: Dictionary) -> bool:
   380				return _json_int(a["season_index"]) < _json_int(b["season_index"])
   381			)
   382			var streak: int = 0
   383			var best_streak: int = 0
   384			for r_v in rows:
   385				var bal: int = _json_int((r_v as Dictionary)["city_balance_li"])
   386				if bal == 0:
   387					streak = streak + 1
   388					if streak > best_streak:
   389						best_streak = streak
   390				else:
   391					streak = 0
   392			if best_streak > longest_zero:
   393				longest_zero = best_streak
   394			if rows.is_empty():
   395				continue
   396			var last: Dictionary = rows[rows.size() - 1]
   397			var last_si: int = _json_int(last["season_index"])
   398			var year_start: int = last_si - seasons_y + 1
   399			var first: Dictionary = last
   400			for r2_v in rows:
   401				var r2: Dictionary = r2_v
   402				if _json_int(r2["season_index"]) == year_start:
   403					first = r2
   404					break
   405			var b0: int = _json_int(first["city_balance_li"])
   406			var b1: int = _json_int(last["city_balance_li"])
   407			var growth: float = 0.0
   408			if b0 != 0:
   409				growth = float(b1) / float(b0)
   410			if growth > worst_growth:
   411				worst_growth = growth
   412		var a_pass := PASS_NA
   413		## D-93／WP-27 MET-05(b): value only, never judge pass (TH-05b cancelled).

   415		if th.has("TH-05a"):

   431	static func _obs01(ports: Array) -> String:
   432		## First season_index per port where city_balance >= any public L2 fee; blank if never.
   433		## Requires BuildingEconomy loaded (public_upgrade_fee assert); else blank per port.
   434		var by_port: Dictionary = _group_ports(ports)
   435		var parts: PackedStringArray = PackedStringArray()
   436		var pids: Array = by_port.keys()
   437		pids.sort()
   438		var be_ready: bool = BuildingEconomy.is_loaded()
   439		for pid_v in pids:
   440			var pid: String = String(pid_v)
   441			var rows: Array = by_port[pid]
   442			rows.sort_custom(func(a: Dictionary, b: Dictionary) -> bool:
   443				return _json_int(a["season_index"]) < _json_int(b["season_index"])
   444			)
   445			var found := ""
   446			if be_ready:
   447				for r_v in rows:
   448					var r: Dictionary = r_v
   449					var fee: Dictionary = BuildingEconomy.public_upgrade_fee(
   450						StringName(pid), BuildingEconomy.BUILDING_LEVEL_MIN + 1
   451					) ## EX-03 first upgrade target L2
   452					var need: int = _json_int(fee.get("CASH_LI", 0))
   453					if need > 0 and _json_int(r["city_balance_li"]) >= need:
   454						found = str(_json_int(r["season_index"]))
   455						break
   456			parts.append("%s:%s" % [pid, found])
   457		return ";".join(parts)
