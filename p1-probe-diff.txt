+static func note_post_explore_depart(sid: String, from_port: String, dest: String) -> void:
+	## Called on first depart after an explore arrival; await publish of origin=D onto dest.
+	if not is_active():
+		return
+	if not _explore_pending.has(sid):
+		return
+	var pe: Dictionary = _explore_pending[sid]
+	if int(pe.get("arrive_day", -1)) < 0:
+		return
+	if int(pe.get("next_depart_day", -1)) >= 0:
+		return
+	pe["next_depart_day"] = _day
+	pe["next_from"] = from_port
+	pe["next_to"] = dest
+	pe["d_port"] = String(pe.get("to", ""))
+	pe["d_on_next_board"] = -1
+	_explore_await_publish[sid] = pe.duplicate(true)
+	_explore_events.append(
+		"NEXT_DEPART day=%d sid=%s from=%s to=%s await_publish_d=%s"
+		% [_day, sid, from_port, dest, String(pe.get("to", ""))]
+	)
+	_explore_pending.erase(sid)

+		int(pe.get("arrive_day", -1)),
+		int(pe.get("sell_ok", -1)),
+		String(pe.get("sell_reason", "")),
+		int(pe.get("next_depart_day", -1)),
+		String(pe.get("next_from", "")),
+		String(pe.get("next_to", "")),
+		int(pe.get("d_on_next_board", -1)),
+	]

+	sid: String, from_port: String, dest: String, _explore: bool
