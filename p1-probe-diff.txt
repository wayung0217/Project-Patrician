+static func _fmt_skip() -> String:
+	var lines: PackedStringArray = PackedStringArray()
+	lines.append("diag_ports_skip_reasons.txt")
+	lines.append("## skip reason counts per from→to (day + season aggregates)")
+	lines.append("scope,day_or_season,from_port,to_port,reason,count")
+	var dkeys: Array = _day_skip.keys()

+			"day,%s,%s,%s,%s,%d"

+	var skeys: Array = _season_skip.keys()

+static func _fmt_route() -> String:
+	var lines: PackedStringArray = PackedStringArray()
+	lines.append("diag_ports_route_matrix.txt")
+	lines.append("## P→D cand_enter + sailed (day + season); season block = 16×16 sail matrix raw")
+	lines.append("scope,day_or_season,from_port,to_port,cand_enter,sailed")
+	var dkeys: Array = _day_route.keys()
+	dkeys.sort()
+	for k in dkeys:
+		var parts: PackedStringArray = String(k).split("|")
+		var row: Dictionary = _day_route[k]
+		lines.append(
+			"day,%s,%s,%s,%d,%d"
+			% [parts[0], parts[1], parts[2], int(row.get("cand_enter", 0)), int(row.get("sailed", 0))]
+		)
+	var skeys: Array = _season_route.keys()
+	skeys.sort()
+	for k in skeys:
+		var parts2: PackedStringArray = String(k).split("|")
+		var row2: Dictionary = _season_route[k]
+		lines.append(
+			"season,%s,%s,%s,%d,%d"
+			% [
+				parts2[0],
+				parts2[1],
+				parts2[2],
+				int(row2.get("cand_enter", 0)),
+				int(row2.get("sailed", 0)),
+			]
+		)
+	## Compact season sail matrices
+	var seasons: Dictionary = {}
+	for k2 in skeys:
+		var p: PackedStringArray = String(k2).split("|")
+		var si: String = p[0]
+		if not seasons.has(si):
+			seasons[si] = []
+		(seasons[si] as Array).append(k2)
+	var sk: Array = seasons.keys()
+	sk.sort()
+	for si2 in sk:
+		lines.append("## MATRIX season=%s (from,to,sailed)" % String(si2))
+		var keys: Array = seasons[si2]
+		keys.sort()
+		for kk in keys:
+			var pp: PackedStringArray = String(kk).split("|")
+			var rr: Dictionary = _season_route[kk]
+			var sailed_n: int = int(rr.get("sailed", 0))
+			if sailed_n > 0:
+				lines.append("%s,%s,%d" % [pp[1], pp[2], sailed_n])
+	return "\n".join(lines) + "\n"
+

+static func _fmt_explore() -> String:
+	var lines: PackedStringArray = PackedStringArray()
+	lines.append("diag_ports_explore.txt")
+	lines.append("## daily explore pick counts + per-event explore lines")
+	lines.append("## EXPLORE_N day,n")
+	var dn: Array = _day_explore_n.keys()
+	dn.sort_custom(func(a, b): return int(a) < int(b))
+	for d in dn:
+		lines.append("EXPLORE_N,%s,%d" % [String(d), int(_day_explore_n[d])])
+	lines.append("## EXPLORE_EVENT lines")
+	for line_v in _explore_lines:
+		lines.append(String(line_v))
+	return "\n".join(lines) + "\n"
