func mid_price(port_id: StringName, good_id: StringName) -> float:
	# FRM-SYS04-03: mid = anchor × m(w); w via single weeks formula (D-34d／N-04).
	assert(_loaded, "MarketPricer.mid_price before load_and_validate")
	_clear_last_error()
	var ctx: Dictionary = _resolve_context(port_id, good_id)
	if ctx.has(KEY_ERROR):
		_fail_from_ctx(ctx)
		return 0.0
	var anchor: int = int(ctx[FIELD_ANCHOR])
	# stock／weekly_offtake written as int／float in _context（C-09: no float→int via int()）.
	var stock_mid: int = ctx["stock"]
	var weekly_mid: float = ctx["weekly_offtake"]
	var w: Variant = _stock_weeks_raw(stock_mid, weekly_mid)
	var m: float = _floor_mult() if w == null else _m_of_w(float(w))
	return float(anchor) * m

func max_affordable_qty(actor: StringName, port: StringName, good: StringName) -> int:

	# q_afford = 最大 q 同時滿足 buy_total(q) ≤ balance 且 q ≤ s
	# ＝ MIN(現金可負擔量, s)；s = 該港該貨 MARKET 庫存。
	# s = 0 → 回 0，不以 q = 1 試探；倍增／二分不得以 q > s 呼叫 quote()
	# （r451：買量須 ≤ s，否則 ERR-SYS04-06；收斂上界＝回到 quote 定義域，非 C-06 clamp）。
	# A18：quote 失敗不得當 total＝0；視為該 q 不可買。
	# 數值例：frm_examples EX-05-09-01 至 03。庫存不再於 FRM-SYS05-10 重做。
	assert(_loaded, "CashLedger.max_affordable_qty before load")
	var bal: int = balance(actor)
	if bal <= 0:
		return 0
	if _quote_provider == null:
		_record_error(ERR_SYS05_04, "quote_provider", null, "quote provider missing (MarketPricer WP-06 gap)")
		return 0
	## FRM-SYS05-09／D-124／D-125 甲（Owner 2026-11-25；v6.99.222）：
	## 庫存上界無條件適用（不限 MarketPricer／DemoQuoteProvider）；

	## （is_loaded 沿用既有 Autoload 慣例）。
	var abs_hi: int = -1
	if GoodsLedger.is_loaded():
		var port_s: String = String(port)
		var market: Dictionary = {
			GoodsLedger.KEY_KIND: GoodsLedger.KIND_MARKET,
			GoodsLedger.KEY_PORT: port_s,
		}
		var stock_cap: int = GoodsLedger.get_qty(market, good)
		## D-124／EX-05-09-03：s = 0 → 0；不得以 q = 1 呼叫 quote()。
		if stock_cap < 1:
			return 0
		abs_hi = stock_cap + 1
	# 擴張上界：hi 倍增直到 total > bal 或 quote 失敗（或碰到 s+1，永不 quote q>s）
	var lo: int = 0
	var hi: int = 1
	while true:
		if abs_hi > 0 and hi >= abs_hi:
			hi = abs_hi
			break
		var t_hi: int = _quote_buy_total(port, good, hi)
		if t_hi < 0 or t_hi > bal:
			break
		lo = hi
		# hi = hi * 2 without literal 2
		var next_hi: int = hi + hi
		if next_hi <= hi:
			# overflow / cap
			break
		if abs_hi > 0 and next_hi >= abs_hi:
			hi = abs_hi
			break
		hi = next_hi
	# 二分
	while lo + 1 < hi:
		var mid: int = floori((lo + hi) / INT_DIV_MID_DENOM)
		var t_mid: int = _quote_buy_total(port, good, mid)
		if t_mid < 0 or t_mid > bal:
			hi = mid
		else:
			lo = mid

	if lo > 0:
		var t_lo: int = _quote_buy_total(port, good, lo)
		if t_lo < 0 or t_lo > bal:

--- Step-through when quote(s) ≤ bal (GoodsLedger loaded, s≥1, provider present) ---

Given: stock_cap = s ≥ 1, abs_hi = s + 1, and for all q in 1..s, _quote_buy_total(port,good,q) ≥ 0 and ≤ bal.
(Premise of H5: in particular quote(s) ≤ bal; buy_total monotone nondecreasing ⇒ all q≤s also ≤ bal.)

Expansion:
  lo=0, hi=1
  Loop: quote(hi) succeeds and ≤ bal → lo=hi; next_hi=hi+hi
  While next_hi < abs_hi: hi=next_hi and continue
  When next_hi ≥ abs_hi: set hi=abs_hi (=s+1), break WITHOUT quoting hi=s+1
  (Never quotes q > s. May never have quoted q=s yet if s is not a power-of-two stop.)

Binary search (lo is last affordable power-of-two ≤ s; hi = s+1):
  Invariant: lo affordable, hi not-yet-proven / exclusive upper (=s+1, never quoted as buy qty)
  while lo + 1 < hi:
    mid = floori((lo + hi) / 2)   ## INT_DIV_MID_DENOM = 2
    ## mid ∈ (lo, hi) ⊂ [1, s]
    if quote(mid) fails or > bal: hi = mid
    else: lo = mid
  Because all q≤s are affordable, every mid ≤ s succeeds → lo climbs to s
  Terminates with lo = s, hi = s + 1

Correction:
  lo = s > 0 → quote(s); by premise ≤ bal → lo unchanged
  return lo = s

Concrete numeric sketch (s=300, abs_hi=301, all quotes ≤ bal):
  expand: 1→2→4→…→256 affordable; next 512≥301 → hi=301, lo=256
  binary mids: 278,289,295,298,299,300 → lo=300
  correct quote(300) ok → return 300 = s

Therefore: when quote(s) ≤ bal (and s≥1, provider ok), current search returns s.
A2 fast-path (single quote(q=s) then return s) matches that result; on failure or total>bal,
fall through to the same expansion/binary/correction (unchanged).
